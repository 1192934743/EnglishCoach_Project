```python
# ============ 纯数据分离：网络 I/O 函数 ============
def _fetch_embeddings_sync(texts_batch: list[tuple[int, str]], embedder) -> dict[int, list[float]]:
    """
    在独立线程中运行：纯同步网络请求 + sleep 限流

    Args:
        texts_batch: [(scenario_id, text), ...] 纯数据结构
        embedder: SyncSemanticEmbedder 实例

    Returns:
        {scenario_id: vector} 纯数据结构
    """
    import time
    vectors_map = {}
    for i in range(0, len(texts_batch), 10):
        chunk = texts_batch[i:i + 10]
        for scenario_id, text in chunk:
            vectors_map[scenario_id] = embedder.embed(text)
        if i + 10 < len(texts_batch):
            time.sleep(1)  # 这里 sleep 只会阻塞当前 worker 线程，非常安全
    return vectors_map


# ============ 异步落库函数 ============
async def ingest_graph_nodes(db, topic, depth, graph_data) -> list[int]:
    """
    将生成的 JSON 数据实例化并写入数据库。

    包含批量 embedding + 限流逻辑。

    架构设计（DTO 模式）：
    1. Step 1: DB I/O - 写入 MicroScenario（主线程，持有 Session）
    2. Step 2: Network I/O - 通过 to_thread 调用（子线程，传递纯数据）
    3. Step 3: DB I/O - 更新 embedding 并写入 Constraints（主线程）
    """
    scenarios_data = graph_data.get("scenarios", [])
    if not scenarios_data:
        logger.warning("LLM 返回了空的场景列表。")
        return []

    # Step 1: 先批量写入 MicroScenario（主线程，持有 DB Session）
    scenarios = []
    for s_data in scenarios_data:
        scenario = MicroScenario(
            topic_id=topic.id,
            scenario_code=s_data["scenario_code"],
            scenario_name=s_data["scenario_name"],
            intent_desc=s_data["intent_desc"],
            scene_desc=s_data["scene_desc"],
            depth_level=depth,
            step_order=s_data["step_order"],
            is_entry_point=s_data["is_entry_point"],
            max_turns=8,
            weight=1.0,
            embedding=None,
        )
        db.add(scenario)
        scenarios.append(scenario)

    db.commit()
    db.refresh(scenarios)

    # Step 2: 构建纯数据结构（只传递 tuple，不传递 ORM 对象）
    texts_batch = []
    for s in scenarios:
        text = f"{s.scenario_name} | Intent: {s.intent_desc} | Scene: {s.scene_desc}"
        texts_batch.append((s.id, text))

    embedder = SyncSemanticEmbedder(cache_enabled=True)

    # 🌟 核心点：将耗时的同步网络请求扔进线程池，且只传递 tuple，完美隔离！
    vectors_map = await asyncio.to_thread(_fetch_embeddings_sync, texts_batch, embedder)

    # Step 3: 更新 embedding（主线程，持有 DB Session）
    for scenario in scenarios:
        scenario.embedding = vectors_map.get(scenario.id)
    db.commit()  # 一次性写入所有 embedding

    # Step 4: 写入 Constraints
    for s_data, scenario in zip(scenarios_data, scenarios):
        constraints_data = s_data.get("constraints", [])
        for c_data in constraints_data:
            constraint = ScenarioConstraint(...)
            db.add(constraint)
    db.commit()

    return [s.id for s in scenarios]
```

### 修改 #4: 导出公共接口 + 事务原子性

**问题描述**:
`graph_generator.py` 的核心生成逻辑无法被 `generate_full_graph.py` 复用，且需要确保事务原子性。

> **关键设计**: 必须先调用 LLM 拿到有效数据，再执行 `clear_existing_graph` 和 `ingest_graph_nodes`，避免旧数据因 LLM 报错而被白白清空。

**修改文件**: `python_backend/scripts/graph_generator.py`

```python
async def generate_and_save(topic: Topic, depth: int, db: Session, rebuild: bool = False) -> list[int]:
    """
    公共接口：生成并保存微场景

    事务原子性保证：
    1. 先调用 LLM 生成数据（可能失败）
    2. 确认数据有效后，再清理旧数据和写入新数据

    Args:
        topic: 话题对象
        depth: 难度层级 (1/2/3)
        db: 数据库会话
        rebuild: 是否先清理旧数据

    Returns:
        新创建的场景 ID 列表
    """
    try:
        # Step 1: 先调用 LLM 生成数据（可能失败）
        graph_data = await generate_micro_scenarios(topic, depth, db)

        # Step 2: 确认数据有效后，清理旧数据
        if rebuild:
            clear_existing_graph(db, topic.id, depth)

        # Step 3: 写入新数据（包含批量 embedding）
        new_ids = ingest_graph_nodes(db, topic, depth, graph_data)
        return new_ids
    except Exception as e:
        db.rollback()  # 失败时必须回滚，防止 PendingRollbackError
        raise
```

### 修改 #6: 异步并发处理（独立 Session）

**问题描述**:
`generate_full_graph.py` 虽有 `RateLimiter`，但实际是串行处理。

> **重要**: SQLAlchemy Session 是非线程安全且非异步安全的，并发时必须每个任务使用独立 Session。

**修改文件**: `generate_full_graph.py`

```python
async def run(self):
    # ... 获取 topics 和 tasks ...

    semaphore = asyncio.Semaphore(self.rate_limiter.max_concurrent)

    async def bounded_generate(topic, depth):
        async with semaphore:
            await self.rate_limiter.acquire()
            # 每个任务使用独立 Session
            with SessionLocal() as task_db:
                try:
                    return await self.generate_topic_scenarios(topic, depth, task_db)
                except Exception as e:
                    task_db.rollback()  # 失败时回滚，不影响其他任务
                    logger.error(f"[ERROR] {topic.title} (Depth {depth}): {e}")
                    return False

    results = await asyncio.gather(
        *[bounded_generate(t, d) for t, d in tasks],
        return_exceptions=True
    )

    # 处理异常结果
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            logger.error(f"[ERROR] Task {i} failed: {result}")
            tasks[i] = False

    success_count = sum(1 for r in results if r is True)
    fail_count = len(results) - success_count
```

def _parse_json_response(text: str) -> dict:
    """解析 LLM 输出，支持从 markdown 中提取 JSON"""
    import json
    import re

    # 直接解析
    try:
        return json.loads(text.strip())
    except json.JSONDecodeError:
        pass

    # 从 markdown 提取
    match = re.search(r'```(?:json)?\s*([\s\S]*?)```', text)
    if match:
        try:
            return json.loads(match.group(1).strip())
        except json.JSONDecodeError:
            pass

    # 通用 JSON 对象提取
    match = re.search(r'\{[\s\S]*\}', text)
    if match:
        try:
            return json.loads(match.group())
        except json.JSONDecodeError:
            pass

    raise ValueError(f"无法解析 LLM 输出: {text[:200]}")

# 在 generate_micro_scenarios 函数中使用

## 五、风险与应对

| 风险 | 影响 | 应对措施 |
|------|------|----------|
| 重构引入 bug | 高 | 每个修改后执行回归测试 |
| 维度修正导致现有数据失效 | 高 | 检查现有数据，需要时执行迁移 |
| 异步并发超出 API 限制 | 中 | RateLimiter + embed_batch 内部限流（chunk_size=10） |
| Session 并发冲突 | 高 | 每个任务使用独立 Session，异常时 rollback |
| LLM 报错导致数据不一致 | 高 | 先 LLM 再 clear，异常时 rollback |

### 5.1 关键设计决策说明

**决策 1: 批量 embedding DTO 模式（彻底隔离网络 I/O 和数据库 I/O）**

```
┌─────────────────────────────────────────────────────────────────────┐
│ 主线程（事件循环）                                                    │
│  ┌──────────────────────────────────────────────────────────────┐   │
│  │ ingest_graph_nodes() [async]                                │   │
│  │                                                              │   │
│  │  Step 1: DB I/O → 写入 MicroScenario（持有 Session）        │   │
│  │              ↓                                              │   │
│  │  Step 2: await asyncio.to_thread(_fetch_embeddings_sync)    │   │
│  │              ↓ 传递 [(id, text), ...] 纯数据                │   │
│  │  Step 3: DB I/O → 更新 embedding + 写入 Constraints          │   │
│  └──────────────────────────────────────────────────────────────┘   │
└─────────────────────────────────────────────────────────────────────┘
                              ↓ 线程边界（纯数据结构）
┌─────────────────────────────────────────────────────────────────────┐
│ 子线程（Worker Thread）                                              │
│  ┌──────────────────────────────────────────────────────────────┐   │
│  │ _fetch_embeddings_sync(texts_batch, embedder)               │   │
│  │                                                              │   │
│  │  for chunk in texts_batch (chunk_size=10):                  │   │
│  │      embedder.embed(text)  ← 同步网络请求                    │   │
│  │      time.sleep(1)         ← 限流（只阻塞当前子线程）        │   │
│  │                                                              │   │
│  │  return {id: vector, ...}  ← 返回纯数据结构                 │   │
│  └──────────────────────────────────────────────────────────────┘   │
└─────────────────────────────────────────────────────────────────────┘
```

**关键要点**:
- 网络 I/O 在子线程中执行，`time.sleep` 不会阻塞事件循环
- 数据库 I/O 在主线程中执行，Session 和 ORM 对象始终在主线程
- 只通过原生 Python 数据结构（tuple, dict）跨越线程边界

**决策 2: 每个并发任务使用独立 Session**

```python
# 正确：独立 Session
async def bounded_generate(topic, depth):
    with SessionLocal() as task_db:
        return await self.generate_topic_scenarios(topic, depth, task_db)
```

**决策 3: 事务原子性**

```python
async def generate_and_save(topic, depth, db, rebuild=False):
    try:
        # Step 1: 先调用 LLM 生成数据（可能失败）
        graph_data = await generate_micro_scenarios(topic, depth, db)

        # Step 2: 确认数据有效后，清理旧数据
        if rebuild:
            clear_existing_graph(db, topic.id, depth)

        # Step 3: 写入新数据（包含批量 embedding）
        new_ids = await ingest_graph_nodes(db, topic, depth, graph_data)
        return new_ids
    except Exception as e:
        db.rollback()  # 失败时必须回滚
        raise
```

> **注意**: `graph_generator.py` 是同步脚本（通过 `asyncio.run()` 调用），`ingest_graph_nodes` 改为 `async def` 后需要使用 `await` 调用。
